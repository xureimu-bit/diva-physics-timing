cmake_minimum_required(VERSION 3.20)
project(DivaPhysicsTiming VERSION 0.1.18 LANGUAGES C CXX ASM_MASM)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "MSVC on Windows x64 is required")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory(source/third_party/minhook)
function(physics_target target)
    target_compile_options(${target} PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4;/WX;/EHsc;/utf-8>)
    target_link_libraries(${target} PRIVATE minhook user32)
endfunction()
add_library(DivaPhysicsTiming SHARED source/DivaPhysicsTiming.cpp source/delta_proxy.asm)
physics_target(DivaPhysicsTiming)
add_library(DivaPhysicsControls MODULE source/controls/DivaPhysicsControls.cpp)
set_target_properties(DivaPhysicsControls PROPERTIES PREFIX "" SUFFIX ".addon64")
target_include_directories(DivaPhysicsControls PRIVATE source/controls/third_party/reshade source/controls/third_party/imgui)
target_compile_definitions(DivaPhysicsControls PRIVATE IMGUI_DISABLE_OBSOLETE_FUNCTIONS)
physics_target(DivaPhysicsControls)
include(CTest)
if(BUILD_TESTING)
    foreach(name core pipeline presentation timing rates settings trace translation rigid roots capture loading parallel delta_abi)
        add_executable(test_${name} source/test_${name}.cpp)
        if(NOT name STREQUAL "core")
            target_sources(test_${name} PRIVATE source/delta_proxy.asm)
        endif()
        if(name STREQUAL "delta_abi")
            target_sources(test_${name} PRIVATE source/test_proxy.asm)
        endif()
        physics_target(test_${name})
        target_compile_options(test_${name} PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/UNDEBUG>)
        add_test(NAME ${name} COMMAND test_${name})
    endforeach()
endif()
option(BUILD_BENCHMARKS "Build optional performance benchmarks" OFF)
if(BUILD_BENCHMARKS)
    foreach(name presentation callbacks parallel)
        add_executable(benchmark_${name} source/benchmark_${name}.cpp source/delta_proxy.asm)
        physics_target(benchmark_${name})
    endforeach()
endif()
install(TARGETS DivaPhysicsTiming RUNTIME DESTINATION "mods/Diva Physics Timing")
install(TARGETS DivaPhysicsControls LIBRARY DESTINATION . RUNTIME DESTINATION .)
install(DIRECTORY dist/ DESTINATION .)
